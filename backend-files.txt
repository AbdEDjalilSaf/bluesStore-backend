<?php

namespace App\Http\Controllers;

use App\Http\Requests\StoreProductRequest;
use App\Http\Requests\UpdateProductRequest;
use App\Http\Resources\ProductResource;
use App\Models\Product;
use App\Support\ProductImageStorage;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;
use Illuminate\Http\UploadedFile;
use Illuminate\Support\Facades\DB;
use OpenApi\Attributes as OA;
use Throwable;

class ProductController extends Controller
{
    /**
     * List all active shirts without pagination or filters.
     */
    #[OA\Get(
        path: '/products',
        summary: 'List all active shirts',
        description: 'Returns all active shirts without pagination, filters, or sorting.',
        tags: ['products'],
        responses: [
            new OA\Response(response: 200, description: 'List of all active shirts.', content: new OA\JsonContent(ref: '#/components/schemas/ProductCollection')),
        ]
    )]
    public function index(Request $request)
    {
        $products = Product::query()
            ->where('is_active', true)
            ->with(['images' => fn ($query) => $query->orderBy('sort_order')])
            ->orderByDesc('created_at')
            ->orderByDesc('id')
            ->get();

        return ProductResource::collection($products);
    }

    /**
     * Create a product with optional images (admin).
     */
    #[OA\Post(
        path: '/products',
        summary: 'Create a product',
        description: 'Create a product and, when images are sent, store them on the public disk and create their rows in the same transaction (admin).',
        tags: ['products'],
        requestBody: new OA\RequestBody(
            required: true,
            content: [
                new OA\JsonContent(ref: '#/components/schemas/ProductRequest'),
                new OA\MediaType(
                    mediaType: 'multipart/form-data',
                    schema: new OA\Schema(ref: '#/components/schemas/ProductRequest')
                ),
            ]
        ),
        responses: [
            new OA\Response(response: 201, description: 'Product created.', content: new OA\JsonContent(ref: '#/components/schemas/Product')),
            new OA\Response(response: 422, description: 'Validation failed.'),
        ]
    )]
    public function store(StoreProductRequest $request)
    {
        $data = $request->validated();
        $files = $data['images'] ?? [];
        unset($data['images']);

        $storedPaths = [];

        try {
            $product = DB::transaction(function () use ($data, $files, &$storedPaths) {
                $product = Product::create($data);

                $this->storeImages($product, $files, $storedPaths);

                return $product;
            });
        } catch (Throwable $exception) {
            ProductImageStorage::delete($storedPaths);

            throw $exception;
        }

        return (new ProductResource($product->load(['images' => fn ($query) => $query->orderBy('sort_order')])))
            ->response()
            ->setStatusCode(201);
    }

    /**
     * Show a single product by its id or slug.
     */
    #[OA\Get(
        path: '/products/{key}',
        summary: 'Show a product',
        description: 'Returns a single active product, resolved by id or slug.',
        tags: ['products'],
        parameters: [
            new OA\Parameter(name: 'key', in: 'path', required: true, description: 'Product id or slug.', schema: new OA\Schema(type: 'string')),
        ],
        responses: [
            new OA\Response(response: 200, description: 'The product.', content: new OA\JsonContent(ref: '#/components/schemas/Product')),
            new OA\Response(response: 404, description: 'Product not found or inactive.'),
        ]
    )]
    public function show(string $key)
    {
        $product = Product::findByKey($key);

        abort_unless($product->is_active, 404);

        return new ProductResource($product->load(['images' => fn ($query) => $query->orderBy('sort_order')]));
    }

    /**
     * Update a product's fields and append or replace uploaded images (admin).
     */
    #[OA\Patch(
        path: '/products/{key}',
        summary: 'Update a product',
        description: 'Update product fields by id or slug. Uploaded images are appended by default; set replace_images to replace the gallery.',
        tags: ['products'],
        parameters: [
            new OA\Parameter(name: 'key', in: 'path', required: true, description: 'Product id or slug.', schema: new OA\Schema(type: 'string')),
        ],
        requestBody: new OA\RequestBody(
            required: true,
            content: [
                new OA\JsonContent(ref: '#/components/schemas/ProductUpdateRequest'),
                new OA\MediaType(
                    mediaType: 'multipart/form-data',
                    schema: new OA\Schema(ref: '#/components/schemas/ProductUpdateRequest')
                ),
            ]
        ),
        responses: [
            new OA\Response(response: 200, description: 'Product updated.', content: new OA\JsonContent(ref: '#/components/schemas/Product')),
            new OA\Response(response: 404, description: 'Product not found.'),
            new OA\Response(response: 422, description: 'Validation failed.'),
        ]
    )]
    public function update(UpdateProductRequest $request, string $key)
    {
        $product = Product::findByKey($key);

        $data = $request->validated();
        $files = $data['images'] ?? [];
        $replaceImages = filter_var($data['replace_images'] ?? false, FILTER_VALIDATE_BOOLEAN);
        unset($data['images'], $data['replace_images']);

        $storedPaths = [];
        $pathsToDelete = [];

        try {
            DB::transaction(function () use ($product, $data, $files, $replaceImages, &$storedPaths, &$pathsToDelete) {
                $product->update($data);

                if ($replaceImages) {
                    $pathsToDelete = $product->images()->pluck('path')->all();
                    $product->images()->delete();
                }

                $this->storeImages($product, $files, $storedPaths);
            });
        } catch (Throwable $exception) {
            ProductImageStorage::delete($storedPaths);

            throw $exception;
        }

        ProductImageStorage::delete($pathsToDelete);

        return new ProductResource($product->load(['images' => fn ($query) => $query->orderBy('sort_order')]));
    }

    /**
     * Delete a product permanently, including its image files (admin).
     */
    #[OA\Delete(
        path: '/products/{key}',
        summary: 'Delete a product',
        description: 'Delete a product by id or slug. Removes its image rows and stored image files (admin).',
        tags: ['products'],
        parameters: [
            new OA\Parameter(name: 'key', in: 'path', required: true, description: 'Product id or slug.', schema: new OA\Schema(type: 'string')),
        ],
        responses: [
            new OA\Response(response: 204, description: 'Product deleted.'),
            new OA\Response(response: 404, description: 'Product not found.'),
        ]
    )]
    public function destroy(string $key): JsonResponse
    {
        $product = Product::findByKey($key);
        $paths = $product->images()->pluck('path')->all();

        $product->delete();

        ProductImageStorage::delete($paths);

        return response()->json(null, 204);
    }

    /**
     * Store the uploaded images on the configured product image provider.
     *
     * @param  array<int, UploadedFile>  $files
     * @param  array<int, string>  $storedPaths
     */
    private function storeImages(Product $product, array $files, array &$storedPaths): void
    {
        if ($files === []) {
            return;
        }

        $lastSortOrder = $product->images()->max('sort_order');
        $nextSortOrder = $lastSortOrder === null ? 0 : ((int) $lastSortOrder) + 1;
        $hasPrimary = $product->images()->where('is_primary', true)->exists();

        foreach (array_values($files) as $index => $file) {
            $path = ProductImageStorage::store($file);

            $storedPaths[] = $path;

            $product->images()->create([
                'path' => $path,
                'sort_order' => $nextSortOrder++,
                'is_primary' => $index === 0 && ! $hasPrimary,
            ]);
        }
    }
}

<?php

namespace App\Support;

use Cloudinary\Cloudinary;
use Illuminate\Http\Request;
use Illuminate\Http\UploadedFile;
use Illuminate\Support\Facades\Log;
use Illuminate\Support\Facades\Storage;
use RuntimeException;
use Throwable;

class ProductImageStorage
{
    private const CLOUDINARY_PREFIX = 'cloudinary:';

    public static function store(UploadedFile $file): string
    {
        $driver = (string) config('filesystems.product_images_driver', 'local');

        if ($driver === 'local') {
            $path = $file->store('products', 'public');

            if ($path === false) {
                throw new RuntimeException('Unable to store the product image.');
            }

            return $path;
        }

        if ($driver !== 'cloudinary') {
            throw new RuntimeException('Unsupported product image storage driver.');
        }

        $realPath = $file->getRealPath();

        if ($realPath === false) {
            throw new RuntimeException('Unable to read the uploaded product image.');
        }

        $result = self::cloudinary()->uploadApi()->upload($realPath, [
            'folder' => 'bluesstore/products',
            'resource_type' => 'image',
            'overwrite' => false,
        ]);

        $publicId = $result['public_id'] ?? null;

        if (! is_string($publicId) || $publicId === '') {
            throw new RuntimeException('Cloudinary did not return a public image ID.');
        }

        return self::CLOUDINARY_PREFIX.$publicId;
    }

    public static function url(string $path, Request $request): string
    {
        if (str_starts_with($path, self::CLOUDINARY_PREFIX)) {
            $publicId = substr($path, strlen(self::CLOUDINARY_PREFIX));

            return (string) self::cloudinary()->image($publicId)->toUrl();
        }

        if (config('filesystems.disks.public.driver') === 'local') {
            $encodedPath = implode('/', array_map('rawurlencode', explode('/', ltrim($path, '/'))));

            return $request->getSchemeAndHttpHost().'/storage/'.$encodedPath;
        }

        return Storage::disk('public')->url($path);
    }

    /**
     * Remove stored files without hiding the original database/upload failure.
     * Cloudinary and local paths can coexist while old product images are migrated.
     *
     * @param  array<int, string>  $paths
     */
    public static function delete(array $paths): void
    {
        foreach ($paths as $path) {
            try {
                if (str_starts_with($path, self::CLOUDINARY_PREFIX)) {
                    self::cloudinary()->uploadApi()->destroy(
                        substr($path, strlen(self::CLOUDINARY_PREFIX)),
                        ['resource_type' => 'image', 'invalidate' => true],
                    );

                    continue;
                }

                Storage::disk('public')->delete($path);
            } catch (Throwable $exception) {
                Log::warning('Unable to delete a product image asset.', [
                    'path' => $path,
                    'error' => $exception->getMessage(),
                ]);
            }
        }
    }

    private static function cloudinary(): Cloudinary
    {
        $cloudinaryUrl = (string) config('services.cloudinary.url', '');

        if ($cloudinaryUrl === '') {
            throw new RuntimeException('CLOUDINARY_URL must be set when PRODUCT_IMAGES_DRIVER=cloudinary.');
        }

        return new Cloudinary($cloudinaryUrl);
    }
}

set -e

# Ensure the framework writable directories exist inside the container.
mkdir -p \
    storage/framework/cache \
    storage/framework/sessions \
    storage/framework/views \
    storage/logs \
    bootstrap/cache

# When using SQLite, make sure the database file exists before migrating.
if [ "${DB_CONNECTION:-sqlite}" = "sqlite" ]; then
    touch "${DB_DATABASE:-database/database.sqlite}"
fi

# Expose the public disk (uploaded product images) under the web root. Force
# repairs a stale/broken symlink after deploy; do not hide failures, otherwise
# the API starts successfully while every uploaded image returns 404.
php artisan storage:link --force

# Apply migrations and seed reference/demo data. The seeders use
# firstOrCreate, so they are safe to run on every boot.
php artisan migrate --force
php artisan db:seed --force
php artisan db:seed --class=DemoProductSeeder --force

exec php artisan serve --host=0.0.0.0 --port="${PORT:-10000}"
